ระบบตั้งเบิกเงินสามารถบันทึกในเบราว์เซอร์ (LocalStorage) ได้ทันทีโดยไม่ต้องตั้งค่าใดๆ แต่หากคุณต้องการให้ซิงค์ข้อมูลบน Cloud กับผู้ใช้อื่นๆ ผ่าน Supabase ให้คัดลอกคำสั่ง SQL ด้านล่างนี้ไปรันใน Supabase Dashboard > SQL Editor: 

CREATE TABLE IF NOT EXISTS disbursements (
  id BIGSERIAL PRIMARY KEY,
  doc_no VARCHAR(50) UNIQUE NOT NULL,
  date DATE NOT NULL,
  requester VARCHAR(100) NOT NULL,
  dept VARCHAR(100) NOT NULL,
  category VARCHAR(100) NOT NULL,
  description TEXT NOT NULL,
  amount NUMERIC(12, 2) NOT NULL DEFAULT 0,
  pay_method VARCHAR(50) DEFAULT 'transfer',
  bank_name VARCHAR(100),
  account_no VARCHAR(100),
  account_name VARCHAR(100),
  receipt_img TEXT,
  note TEXT,
  status VARCHAR(30) DEFAULT 'pending',
  approved_by VARCHAR(100),
  approved_at TIMESTAMPTZ,
  paid_by VARCHAR(100),
  paid_at TIMESTAMPTZ,
  paid_ref VARCHAR(100),
  rejected_reason TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- เปิด Row Level Security (RLS) และอนุญาตการเข้าถึง
ALTER TABLE disbursements ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow all operations for anon" ON disbursements FOR ALL USING (true) WITH CHECK (true);
